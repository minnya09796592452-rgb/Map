	private void CheckElements(IList<TElement> elements, Vector3 position)
	{
		float current = Angle(new Vector2(position.x, position.y));
		float angleStep = GetAngleStep();
		foreach (TElement element in elements)
		{
			float target = Angle(element.transform.position - transform.position);
			if (Mathf.Abs(Mathf.DeltaAngle(current, target)) < angleStep / 2f * _angleThreshold / 100f && _active && !_animating)
			{
				if (_currentElement != element)
				{
					if (!_currentElement.IsNullOrDestroyed())
					{
						Element_Exit(_currentElement);
					}
					_currentElement = element;
					Element_Enter(element);
				}
			}
			else if (_currentElement == element)
			{
				Element_Exit(element);
				_currentElement = null;
			}
		}
	}

			float y = Mathf.Cos(angleForIndex);
			transform.transform.localPosition = new Vector3(x, y, 0f) * _elementRadius;
			transform.transform.localEulerAngles = new Vector3(0f, 0f, (float)_random.Next() * 360f);
		}
	}

			return;

		int i = 0;
		int num = 0;
		for (; i < _elementLayoutCount; i++)
		{
			if (!_reservedIndexes.Contains(i))
			{
				TElement val = PopElement();
				float angleForIndex = GetAngleForIndex(i);
				float x = Mathf.Sin(angleForIndex);
				float y = Mathf.Cos(angleForIndex);
				val.transform.localPosition = new Vector3(x, y, 0f) * _elementRadius;
				val.transform.localEulerAngles = new Vector3(0f, 0f, 0f - rotationOffsetDeg);
				val.Initialize(_models[num]);
				_activeElements.Add(val);
				num++;
				if (num > _models.Count - 1)
				{
					break;
				}
			}
		}
		_view.ElementContainer.Transform.localEulerAngles = new Vector3(0f, 0f, rotationOffsetDeg);
	}

	private float GetAngleStep()
	{
		if (_elementsDistributionType == EElementDistributionType.Even)
		{
			return 360f / (float)_elementLayoutCount;
		}
		return _angleSteps;
	}

	private float GetAngleForIndex(int index)
	{
		if (_elementsDistributionType == EElementDistributionType.Even)
		{
			return MathF.PI * 2f / (float)_elementLayoutCount * (float)index;
		}
		return GetAngleStep() * (MathF.PI / 180f) * (float)index;
	}

	private float GetRotationOffsetDeg()
	{
		return _elementsDistributionType switch
		{
			EElementDistributionType.Even => 0f, 
			EElementDistributionType.FanAroundNorth => (float)(_elementLayoutCount - 1) * GetAngleStep() / 2f, 
			_ => _rotationAngle, 
		};
	}

	private Vector3 GetConstrainedPosition(Vector3 midPoint, Vector3 endPoint)
	{
		float magnitude = (midPoint - endPoint).magnitude;
		_active = magnitude > _selectorRadius * _activationPercentage;
		if (magnitude > _selectorRadius)
		{
			return (endPoint - midPoint).normalized * _selectorRadius + midPoint;
		}
		return endPoint;
	}

	private float Angle(Vector2 angle)
	{
		if (angle.x < 0f)
		{
			return 360f - Mathf.Atan2(angle.x, angle.y) * 57.29578f * -1f;
		}
		return Mathf.Atan2(angle.x, angle.y) * 57.29578f;
	}

		return leaderboardEntryElement;
	}

	public override void CreateLeaderboardEntries(IEnumerable<LeaderboardEntryData> leaderboardEntries)
	{
		BeamTeam.DestroyAllChildrenImmediate(_leaderboardEntryElementContainer);
		int num = 1;
		foreach (LeaderboardEntryData leaderboardEntry in leaderboardEntries)
		{
			LeaderboardEntryElement leaderboardEntryElement = CreateLeaderboardEntryElement();
			PopulateLeaderboardEntryElement(leaderboardEntryElement, leaderboardEntry);
			leaderboardEntryElement.SetRank(num++);
		}
		bool flag = _leaderboardEntryElementContainer.childCount > 0;
		_leaderboardEntryElementContainer.gameObject.SetActive(flag);
		_noLeaderboardEntriesLabel.transform.gameObject.SetActive(!flag);
		_retrievingLabel.transform.gameObject.SetActive(value: false);
	}

	public override void SetLeaderboard(LeaderboardData leaderboardData)
	{
		_leaderboardDisplayType = leaderboardData.DisplayType;
		_leaderboardEntryElementContainer.gameObject.SetActive(value: false);
		_noLeaderboardEntriesLabel.transform.gameObject.SetActive(value: false);
		_retrievingLabel.transform.gameObject.SetActive(value: true);
	}

	public override void SetUserLeaderboardEntry(LeaderboardEntryData leaderboardEntryData)
	{
		PopulateLeaderboardEntryElement(_userLeaderboardEntryElement, leaderboardEntryData);
	}

	private void PopulateLeaderboardEntryElement(LeaderboardEntryElement leaderboardEntryElement, LeaderboardEntryData leaderboardEntryData)
	{
		leaderboardEntryElement.SetDisplayType(_leaderboardDisplayType);
		leaderboardEntryElement.SetUsername(leaderboardEntryData.Username);
		leaderboardEntryElement.SetRank(leaderboardEntryData.Rank);
		leaderboardEntryElement.SetValue(leaderboardEntryData.Value);
	}

	public override IButtonViewAdapter GetFirstButton()
	{
		IButtonViewAdapter buttonViewAdapter = _leaderboardButtons.FirstOrDefault_NonAlloc();
		if (buttonViewAdapter.IsNullOrDestroyed())
		{
			return _backButton;
		}
		return buttonViewAdapter;
	}

	private void LeaderboardButton_Selected(ISelectableViewAdapter widget)
	{
		_scrollView.FocusElement(widget.transform);
	}
