		if (_fuel > 0f)
		{
			if (!AnchorsDeployed() && !Rigidbody.isKinematic)
			{
				Vector3 vector = transform.forward * Mathf.Cos(num) - Rigidbody.transform.right * Mathf.Sin(num);
				Rigidbody.AddForce(vector * (_engineForce * Throttle));
			}
		}
		else if (!_ranOutOfFuel)
		{
			_fuel = 0f;
			_ranOutOfFuel = true;
			OnFuelRanOut();
		}
		if (!Rigidbody.isKinematic)
		{
			float num2 = Vector3.Dot(Rigidbody.velocity, transform.forward);
			Rigidbody.AddTorque(transform.up * (num2 * _rudderDeflection * _rudderInput));
		}
		OperatingPosition.transform.rotation = Operator.Character.PlayerCharacterPosition.rotation;
		OnUsing();
	}

			Vector3 tangent = new Vector3(groundNormal.x, 0f - groundNormal.y, groundNormal.z);
			Vector3.OrthoNormalize(ref groundNormal, ref tangent);
			_targetVelocity = tangent;
			_targetVelocity *= slideSpeed;
		}
		else
		{
			CheckSpeed();
			_targetVelocity = new Vector3(_inputX * _inputModifyFactor, 0f - _antiBumpFactor, _inputY * _inputModifyFactor);
			_targetVelocity = transform.TransformDirection(_targetVelocity);
			_targetVelocity *= _playerSpeed;
			if (groundNormal.y != 0f)
			{
				_targetVelocity *= Mathf.Clamp01(groundNormal.y);
			}
		}
		if (!IsBusy)
		{
			DoCrouch();
			DoSprint();
			CheckJump();
		}
	}

			}
			else
			{
				num2 = (isFloating ? swimSpeed : diveSpeed);
			}
			playerSpeed2 = num2;
		}
		if (CheckBrokenBones())
		{
			float playerSpeed3 = _playerSpeed * 0.5f;
			_playerSpeed = playerSpeed3;
		}
		if (isFloating)
		{
			Vector3 direction = new Vector3(_inputX * _inputModifyFactor, 0f, _inputY * _inputModifyFactor);
			direction = transform.TransformDirection(direction);
			direction *= _playerSpeed;
			CheckAndIncreasePhysicalSkills(direction);
			if ((_player.PlayerCamera.MouseLook.RotationY > 25f && (double)_inputY > 0.0) || _inAirTime > 1f || _jumpCooldownTimer < _jumpInterval)
			{
				direction = new Vector3(_inputX * _inputModifyFactor, 0f, _inputY * _inputModifyFactor);
				direction = _player.PlayerCamera.Camera.transform.TransformDirection(direction);
				direction *= _playerSpeed;
			}
			else
			{
				_inAirTime = 0f;
				int num3 = 20;
				float num4 = _oceanHeight - transform.position.y;
				direction.y = num4 * (float)num3;
				_targetVelocity.y = num4 * (float)num3;
			}
			_targetVelocity = Vector3.Lerp(_targetVelocity, direction, DeltaTime * 1.2f);
		}
		if (!isFloating)
		{
			_inAirTime = 0f;
			Vector3 direction2 = new Vector3(_inputX * _inputModifyFactor, 0f, _inputY * _inputModifyFactor);
			direction2 = _player.PlayerCamera.Camera.transform.TransformDirection(direction2);
			if (_inputClimb)
			{
				direction2.y += 0.5f;
			}
			else if (_inputCrouch)
			{
				direction2.y -= 0.5f;
			}
			direction2 *= _playerSpeed;
			float num5 = Mathf.Clamp(Mathf.Abs(_characterController.velocity.magnitude) * 1.5f, 1f, 20f);
			_targetVelocity = Vector3.Lerp(_targetVelocity, direction2, DeltaTime * num5);
			CheckAndIncreasePhysicalSkills(direction2);
		}
		if (!IsBusy)
		{
			CheckJump();
		}
		if (falling)
		{
			falling = false;
			DoLand();
		}
	}

	private void DoFall()
	{
		_inAirTime += DeltaTime;
		if (!falling && _inAirTime > 0.3f)
		{
			falling = true;
		}
		if (!sliding && !IsBusy)
		{
			float num = _playerSpeed * 0.65f;
			_targetVelocity.x = _inputX * num * _inputModifyFactor;
			_targetVelocity.z = _inputY * num * _inputModifyFactor;
			_targetVelocity = transform.TransformDirection(_targetVelocity);
		}
		if (CheckCollisionUp() && _targetVelocity.y > 0f)
		{
			_targetVelocity.y = _gravity;
		}
	}

	private void DoJump()
	{
		_player.Character.Jump();
		crouched = false;
		if (CheckBrokenBones())
		{
			_player.Statistics.ReplicatedApplyDamage(10f, transform.position + Vector3.down, "Fracture");
		}
		_extraJumpVelocity = Vector3.zero;
		if (ActivePlatform != null && ActivePlatform.GetComponent<Rigidbody>() != null)
		{
			_extraJumpVelocity = ActivePlatform.GetComponent<Rigidbody>().velocity;
		}
		if (_climbingLadder)
		{
			if (!CheckCollisionUp())
			{
				_targetVelocity.y = _jumpHeight;
			}
		}
		else if (InWater)
		{
			_targetVelocity.y = _jumpHeight * 0.75f;
		}
		else
		{
			_targetVelocity.y = _jumpHeight;
		}
		OnJumped();
	}

	private void DoLand()
	{
		_player.SetVibration(0, 0.08f, 0.1f);
		_extraJumpVelocity = Vector3.zero;
		if (_fallStartLevel < transform.position.y)
		{
			return;
		}
		_player.Character.Fall_Land();
		if (InWater)
		{
			OnLanded(LandType.Water);
		}
		else
		{
			if (_climbingLedge || _climbingLadder)
			{
				return;
			}
			float num = 4f;
			float num2 = _fallStartLevel - transform.position.y;
			float damage = num2 * 25f;
			if (num2 > num)
			{
				if (num2 > 6f)
				{
					ReplicatedBreakBones();
				}
				_player.Statistics.ReplicatedApplyDamage(damage, "Falling");
			}
			else
			{
				_ = 1f;
			}
			OnLanded(LandType.Ground);
		}
	}

	private void DoClimbLadder()
	{
		if (_climbingLadder)
		{
			if ((grounded || InWater) && _inputY < 0f)
			{
				activeLadder = null;
			}
			grounded = true;
		}
		if (activeLadder == null)
		{
			ExitLadder();
			return;
		}
		if (!_climbingLadder)
		{
			OnBeginClimbing();
		}
		_climbingLadder = true;
		crouched = false;
		_inAirTime = 0f;
		sprinting = false;
		BoxCollider ladderCollider = activeLadder.LadderCollider;
		if (!_player.PlayerCamera.MouseLook.ClampLadder)
		{
			if (activeLadder.LandPos != null && transform.position.y > activeLadder.LandPos.position.y && !_climbingLedge)
			{
				transform.position = new Vector3(transform.position.x, activeLadder.LandPos.position.y - 0.5f, transform.position.z);
			}
			float num = Vector3.Distance(transform.position, activeLadder.transform.position);
			Vector3 vector = activeLadder.transform.position + activeLadder.transform.up * num;
			Vector3 vector2 = vector + new Vector3(ladderCollider.center.x, 0f, ladderCollider.center.z);
			Vector3 vector3 = new Vector3(vector2.x, transform.position.y, vector2.z) - transform.position;
			vector3.Normalize();
			if (!activeLadder.isTree)
			{
				vector3 = activeLadder.transform.forward;
			}
			_ladderRot = Quaternion.FromToRotation(transform.forward, vector3) * transform.rotation;
			_controllerRotation = _ladderRot;
			transform.rotation = _controllerRotation;
			if (activeLadder.isTree)
			{
				float num2 = Vector3.Distance(new Vector3(vector.x, transform.position.y, vector.z), transform.position);
				if (num2 > _treeSnapOffset)
				{
					transform.position += transform.forward * (num2 - _treeSnapOffset);
				}
				else
				{
					transform.position += -transform.forward * (_treeSnapOffset - num2);
				}
			}
			else
			{
				transform.position = new Vector3(activeLadder.transform.position.x, transform.position.y, activeLadder.transform.position.z) + -vector3 * _ladderSnapOffset;
			}
			_player.Character.CurrentCharacterPosition = _player.Character.LadderHelperPosition;
			_player.Character.CurrentCharacterPosition.up = activeLadder.transform.up;
			_player.Character.CurrentCharacterPosition.rotation = _player.Character.CurrentCharacterPosition.rotation * _ladderRot;
			_player.PlayerCamera.MouseLook.ClampLadder = true;
			_player.PlayerCamera.MouseLook.ClampViewToPos(vector3);
		}
		_controllerRotation = _ladderRot;
		_playerSpeed = ladderClimbSpeed;
		_targetVelocity = _player.Character.CurrentCharacterPosition.up * (_inputY * _inputModifyFactor * _playerSpeed);
		if (activeLadder.isTree)
		{
			Vector3 vector4 = activeLadder.transform.position + activeLadder.transform.up * ladderCollider.size.y;
			if (transform.position.y >= vector4.y - _climbOffsetTreeEnd && _inputY > 0f)
			{
				_targetVelocity = Vector3.zero;
			}
		}
		else if (activeLadder.LandPos != null && _inputY > 0f && Mathf.Abs(transform.position.y - activeLadder.LandPos.position.y) < 0.8f && !_climbingLedge)
		{
			ClimbLedge(activeLadder.LandPos.position + new Vector3(0f, 0.9f, 0f) + -activeLadder.transform.forward * 0.45f);
			return;
		}
		if (!InWater && !IsBusy && CheckJump())
		{
			transform.position += -_player.Character.CurrentCharacterPosition.forward * (_treeSnapOffset * 2f);
			_playerSpeed = 4f;
			ExitLadder();
		}
		if (falling)
		{
			falling = false;
			DoLand();
		}
	}

	private void DoClimbLedge()
	{
		if (_climbingLedge)
		{
			_inAirTime = 0f;
			_targetVelocity = Vector3.zero;
		}
		else if (_canClimbLedge && _inputClimbDownTime > 0.3f)
		{
			ClimbLedge(activeLedge.LandPos.position + new Vector3(0f, 0.9f, 0f) + activeLedge.transform.forward * 0.45f, activeLedge);
		}
	}

	private void ClimbLedge(Vector3 climbPosition, Player_Ledge ledge = null)
	{
		if (!_climbingLedge)
		{
			_ledgeClimbed = ledge;
			sprinting = false;
			Vector3 rotationAngle = Vector3.zero;
			if (_ledgeClimbed != null && _ledgeClimbed.IsRaft)
			{
				_player.PlayerCamera.transform.LookAt(_ledgeClimbed.transform.position);
				_player.PlayerCamera.MouseLook.transform.LookAt(_ledgeClimbed.transform.position);
				Vector3 vector = new Vector3(transform.position.x, _ledgeClimbed.transform.position.y, transform.position.z);
				rotationAngle = _ledgeClimbed.transform.position - vector;
				float num = 0.1f + _ledgeClimbed.GetComponent<BoxCollider>().size.y / 2f;
				Vector3 vector2 = Vector3.MoveTowards(vector, _ledgeClimbed.transform.position, 0.45f) + Vector3.up * num;
				Vector3 vector3 = new Vector3(0f, 0.7f, 0f) + _ledgeClimbed.transform.forward * 0.45f;
				climbPosition = vector2 + vector3;
			}
			else if (_ledgeClimbed != null)
			{
				_player.PlayerCamera.transform.forward = -_ledgeClimbed.transform.forward;
				_player.PlayerCamera.MouseLook.transform.forward = -_ledgeClimbed.transform.forward;
				rotationAngle = -_ledgeClimbed.transform.forward;
			}
			else if (activeLadder != null)
			{
				_player.PlayerCamera.transform.forward = activeLadder.transform.forward;
				_player.PlayerCamera.MouseLook.transform.forward = activeLadder.transform.forward;
				rotationAngle = activeLadder.transform.forward;
			}
			rotationAngle.Normalize();
			_player.PlayerCamera.MouseLook.ClampLedge = true;
			_player.PlayerCamera.MouseLook.ClampViewToPos(rotationAngle);
			_characterController.enabled = false;
			_targetVelocity = Vector3.zero;
			++IsBusy;
			_climbingLedge = true;
			crouched = false;
			InWater = false;
			isFloating = false;
			_player.Character.StartCoroutine(_player.Character.ClimbAnimation());
			transform.position = climbPosition;
			OnClimbedLedge();
		}
	}

	public void FinishLedgeClimb()
	{
		_ = Vector3.zero;
		Vector3 vector = Vector3.up * (_characterController.height / 2f);
		if (activeLadder != null && _climbingLadder)
		{
			if (activeLadder.LandPos != null)
			{
				transform.position = activeLadder.LandPos.position + vector;
			}
			else
			{
				Debug.LogError("Movement :: Unassigned Ladder Ledge 'Landing Position'.");
			}
			ExitLadder();
		}
		else if (_ledgeClimbed != null)
		{
			if (_ledgeClimbed.LandPos != null)
			{
				transform.position = _ledgeClimbed.LandPos.position + vector;
			}
			else
			{
				Debug.LogError("Movement :: Unassigned Ledge 'Landing Position'.");
			}
			_ledgeClimbed = null;
		}
		_player.PlayerCamera.MouseLook.ClampLedge = false;
		_characterController.enabled = true;
		_targetVelocity = Vector3.zero;
		--IsBusy;
		_climbingLedge = false;
		activeLedge = null;
	}

	private void CheckControllerHeight()
	{
		if (_climbingLedge)
		{
			if (_characterController.height != _playerControllerHeight)
			{
				SetControllerHeight(_playerControllerHeight);
			}
		}
		else if (crouched || Vehicle != null)
		{
			if (_characterController.height != _crouchControllerHeight)
			{
				SetControllerHeight(_crouchControllerHeight);
			}
		}
		else if (InWater)
		{
			if (_characterController.height != _swimHeight)
			{

	private bool CheckJump()
	{
		if (!canJump)
		{
			return false;
		}
		if (_climbingLedge)
		{
			return false;
		}
		if (_jumpCooldownTimer >= _jumpInterval)
		{
			bool flag = _inputJumpPress;
			if (_canClimbLedge || (_canClimbLadder && !_climbingLadder))
			{
				flag = _inputJumpPressUp;
			}
			if (flag && (!CheckCollisionUp() || _climbingLadder))
			{
				if (!InWater || !isFloating)
				{
					_jumpCooldownTimer = 0f;
					DoJump();
					return true;
				}
				if (Physics.SphereCast(new Ray(_player.PlayerCamera.transform.position + Vector3.up * 1.5f, Vector3.down), _characterController.radius * 2f, _playerControllerHeight + 1.5f, clipMask.value, QueryTriggerInteraction.Ignore))
				{
					_jumpCooldownTimer = 0f;
					DoJump();
					return true;
				}
			}
		}
		else
		{
			_jumpCooldownTimer += DeltaTime;
		}
		return false;
	}

	private void CheckSliding()
	{
		Vector3 tangent = new Vector3(groundNormal.x, 0f - groundNormal.y, groundNormal.z);
		Vector3.OrthoNormalize(ref groundNormal, ref tangent);
		if (TooSteep() && groundNormal.y != 0f && calculatedVelocity.magnitude > 0.2f && !Physics.Raycast(transform.position, tangent, _characterController.height * 0.5f, clipMask.value, QueryTriggerInteraction.Ignore))
		{
			sliding = true;
		}
		if (_debug && groundNormal.y != 0f)
		{
			Debug.DrawRay(transform.position, _targetVelocity, Color.red);
		}
	}

	private void CheckFalling()
	{
		if (!falling && (grounded || _climbingLadder || _climbingLedge || InWater))
		{
			_fallStartLevel = transform.position.y;
		}
	}

	private void CheckLedge()
	{
		_canClimbLedge = false;
		if (!(activeLedge == null) && !_climbingLedge && !IsUnderwater && (activeLedge.LandPos.position + activeLedge.LandPos.up).y > transform.position.y)
		{
			Vector3 forward = transform.forward;
			forward.y = 0f;
			forward.Normalize();
			Vector3 rhs = transform.position - activeLedge.LandPos.position;
			rhs.y = 0f;
			rhs.Normalize();
			if (Vector3.Dot(forward, rhs) < -0.2f && Vector3.Dot(activeLedge.transform.up, Vector3.up) > 0.9f)
			{
				_canClimbLedge = true;
			}
		}
	}

	private void CheckLadder()
	{
		if (activeLadder == null && !_climbingLadder && canJump)
		{
			_player.PlayerCamera.MouseLook.ClampLadder = false;
		}
		_canClimbLadder = false;
		if ((bool)activeLadder && !crouched)
		{
			if (_climbingLadder)
			{
				_canClimbLadder = true;
			}
			else
			{
				float num = Vector3.Distance(transform.position, activeLadder.transform.position);
				Vector3 vector = activeLadder.transform.position + activeLadder.transform.up * num;
				Vector3 forward = transform.forward;
				forward.y = 0f;
				forward.Normalize();
				Vector3 rhs = transform.position - vector;
				rhs.y = 0f;
				rhs.Normalize();
				if (activeLadder.transform.up.y > 0.7f && Vector3.Dot(forward, rhs) < -0.75f)
				{
					_canClimbLadder = true;
				}
			}
		}
		if (!_canClimbLadder)
		{
			_climbingLadder = false;
		}
	}

	private void CheckMovingPlatform()
	{
		if ((bool)ActivePlatform && _previousPlatform != ActivePlatform)
		{
			_previousPlatform = ActivePlatform.GetComponent<MovingPlatform>();
		}
		bool activePlatformFollow = false;
		Transform activePlatform = null;
		_dragger.CanDrag = true;
		if (grounded && Physics.SphereCast(PlayerPosition, _characterController.radius, -base.transform.up, out var hitInfo, 2f, clipMask.value, QueryTriggerInteraction.Ignore) && !_climbingLedge && !_climbingLadder)
		{
			if (hitInfo.collider.gameObject.CompareTag("Terrain") && !InWater)
			{
				terrainGrounded = true;
				if (!_reachedShoreCheck)
				{
					_reachedShoreCheck = true;
					_onReachedShore?.Invoke();
					_onReachedShore = null;
				}
			}
			else if ((bool)hitInfo.rigidbody)
			{
				if (hitInfo.rigidbody.CompareTag("Raft"))
				{
					Transform transform = hitInfo.rigidbody.transform;
					MovingPlatform component = transform.GetComponent<MovingPlatform>();
					activePlatform = transform;
					activePlatformFollow = true;
					if (!component.Calibrated)
					{
						component.OnEnter();
					}
				}
				if (_dragger.Draggable != null && _dragger.Draggable.rigidbody == hitInfo.rigidbody)
				{
					_dragger.CanDrag = false;
				}
			}
		}
		ActivePlatform = activePlatform;
		ActivePlatformFollow = activePlatformFollow;
		if (ActivePlatform == null && (bool)_previousPlatform && _previousPlatform != ActivePlatform)
		{
			_previousPlatform.OnExit();
			_previousPlatform = null;
		}
	}

	private void CheckInWater()
	{
		float num = transform.position.y + _characterController.center.y;
		bool flag = _player.PlayerCamera.transform.position.y <= _oceanHeight;
		bool flag2 = grounded;
		if (CheckCollisionDown())
		{
			flag2 = true;
		}
		bool inWater = false;
		if (num <= _oceanHeight)
		{
			inWater = true;
		}
		else if (flag2 && !flag)
		{
			inWater = false;
		}
		InWater = inWater;
		isFloating = false;
		if (InWater)
		{
			if (!flag)
			{
				isFloating = true;
			}
			if (_player.PlayerCamera.MouseLook.RotationY > 25f && (double)_inputY > 0.0)
			{
				isFloating = false;
			}
		}
		touchingWater = transform.position.y - _playerControllerHeight * 0.5f < _oceanHeight;
	}

	private void CheckOceanHeight()
	{
		_oceanHeight = Ceto.Ocean.Instance.QueryWaves(transform.position.x, transform.position.z) + _playerFloatOffset;
		_oceanHeight += Mathf.Cos(Time.time * 2f) * 0.1f;
	}

	private bool CheckCollisionUp()
	{
		return Physics.SphereCast(new Ray(PlayerPosition, transform.up), _characterController.radius, _ccRayDistance - _characterController.radius, clipMask.value, QueryTriggerInteraction.Ignore);
	}

	private bool CheckCollisionDown()
	{
		groundedAnimal = false;
		if (Physics.Raycast(new Ray(PlayerPosition, -transform.up), out var hitInfo, 1f, 1 << Layers.TERRAIN_OBJECTS) && hitInfo.collider.gameObject.CompareTag("Animal"))
		{
			groundedAnimal = true;
		}
		groundedAnimal = grounded && groundedAnimal;
		return Physics.SphereCast(new Ray(PlayerPosition, -transform.up), _characterController.radius, _ccRayDistance - _characterController.radius, clipMask.value, QueryTriggerInteraction.Ignore);
	}

	private bool CheckBrokenBones()
	{
		return _player.Statistics.HasStatusEffect(StatusEffect.BROKEN_BONES);
	}

				physicalSkillCheckTimer = 1f;

			_player.PlayerSkills.ReplicatedAddPhysical(_skillExpSource);

	}

	public float GetGroundPosition(Vector3 pos, bool controllerOffset = false)
	{
		Vector3 vector = pos;
		if (Physics.SphereCast(pos, _characterController.radius, -transform.up, out var hitInfo, 2f, clipMask.value, QueryTriggerInteraction.Ignore))

			vector = hitInfo.point;

	private void FixedUpdate()
	{
		if (!_initialized || LevelLoader.IsLoadingScene() || !_poll || Time.timeScale == 0f)
		{
			return;
		}
		_velocity = Mathf.Clamp(_controller.velocity.magnitude, 0f, 20f);
		_inWater = _player.Movement.InWater;
		_extremeCalorieBurn = _player.Movement.sprinting;
		if (Invincible)
		{
			_player.Movement.CanSprint = true;
		}
		else
		{
			if (_isKnockedOut)
			{
				return;
			}
			float num = 0f;
			bool healthy = true;
			IEnumerator<PlayerEffect> enumerator = _playerEffects.GetEnumerator();
			while (enumerator.MoveNext())
			{
				PlayerEffect current = enumerator.Current;
				if (current.Duration != -1f && GameTime.GetHoursSince(current.StartedTime) > current.Duration)
				{
					RemoveStatusEffect(current);
					return;
				}
				_health += current.HealthPerHour * GameTime.DeltaTimeHour;
				_fluids += current.FluidsPerHour * GameTime.DeltaTimeHour;
				_calories += current.CaloriesPerHour * GameTime.DeltaTimeHour;
				num += current.TemperaturePerHour * GameTime.DeltaTimeHour;
				if (!current.PositiveEffect)
				{
					healthy = false;
				}
				if (_health <= 0f)
				{
					ReplicatedKnockout();
					return;
				}
			}
			enumerator.Dispose();
			UpdateVitals(healthy);
			UpdateServerClientCopy();
		}
	}

	private void UpdateVitals(bool healthy = true)
	{
		UpdatePlayerHunger();
		UpdatePlayerThirst();
		UpdateOxygen();
		UpdatePlayerHealth(healthy);
		UpdatePlayerSleep();
		UpdateStamina();
	}

	private void UpdatePlayerHealth(bool healthy)
	{
		if (_health <= 0f)
		{
			ReplicatedKnockout();
			return;
		}
		if (healthy && _calories >= 500f && _fluids >= 500f)
		{
			ApplyStatusEffect(StatusEffect.HEALTH_REGENERATION);
		}
		else
		{
			RemoveStatusEffect(StatusEffect.HEALTH_REGENERATION);
		}
		_health = Mathf.Clamp(_health, 0f, 700f);
	}

	private void UpdatePlayerHunger()
	{
		float num = (_extremeCalorieBurn ? 1.2f : 1f);
		_calorieBurnFactor = CALORIE_POINTS_PER_DAY * num;
		_nutrientBurnFactor = NUTRIENT_POINTS_PER_DAY * num;
		_calories -= _calorieBurnFactor * GameTime.DeltaTimeDay;
		_calories = Mathf.Clamp(_calories, 0f, 700f);
		if (Mathf.Approximately(_calories, 0f))
		{
			ApplyStatusEffect(StatusEffect.STARVATION);
		}
		else
		{
			RemoveStatusEffect(StatusEffect.STARVATION);
		}
		if (_fibres > 0f && _fats > 0f && _proteins > 0f)
		{
			RemoveStatusEffect(StatusEffect.MALNUTRITION);
		}
		else
		{
			ApplyStatusEffect(StatusEffect.MALNUTRITION);
		}
	}

	private void UpdatePlayerThirst()
	{
		float num = (_extremeCalorieBurn ? 1.2f : 1f);
		_calorieBurnFactor = THIRST_POINTS_PER_DAY * num;
		_fluids -= GameTime.DeltaTimeDay * _calorieBurnFactor;
		_fluids = Mathf.Clamp(_fluids, 0f, 700f);
		if (Mathf.Approximately(_fluids, 0f))
		{
			ApplyStatusEffect(StatusEffect.DEHYDRATION);
		}
		else
		{
			RemoveStatusEffect(StatusEffect.DEHYDRATION);
		}
	}

	private void UpdatePlayerSleep()
	{
		_player.Movement.CanSprint = true;
	}

	private void UpdateOxygen()
	{
		bool flag = Cameras.IsUnderwater(_player.PlayerCamera.transform);
		if (_isKnockedOut)
		{
			flag = false;
		}
		if (!flag)
		{
			if (_oxygen < _maxOxygen * _player.PlayerSkills.PhysicalMultiplier)
			{
				_oxygen += GameTime.RealDeltaTime * _oxygenRegainSpeed;
			}
			if (_underWaterTime > 0f)
			{
				OnResurfaced(_underWaterTime);
				_underWaterTime = 0f;
			}
		}
		else
		{
			if (Mathf.Approximately(_underWaterTime, 0f))
			{
				OnDived();
			}
			_underWaterTime += GameTime.RealDeltaTime;
			_oxygenRestored = false;
			if (_oxygen > 0f)
			{
				bool flag2 = HasStatusEffect(StatusEffect.BOOST_BREATH);
				_oxygen -= GameTime.RealDeltaTime * (_oxygenDrainSpeed * (flag2 ? 0.5f : 1f));
			}
		}
		if (_oxygen <= 0f)
		{
			_drowningRestored = false;
			_drownTimer += GameTime.RealDeltaTime;
			if (_drownTimer > _drownTime)
			{
				ReplicatedKnockout();
			}
			ApplyStatusEffect(StatusEffect.DROWNING);
		}
		else
		{
			_drownTimer = 0f;
			RemoveStatusEffect(StatusEffect.DROWNING);
		}
		_drownTimer = Mathf.Clamp(_drownTimer, 0f, _drownTime);
		if (_drownTimer > 0f && !Mathf.Approximately(_drownTimer, _drownTime))
		{
			float value = _drownTimer / _drownTime;
			OnDrowningUpdated(value);
		}
		else if (!_drowningRestored)
		{
			OnDrowningRestored();
			_drowningRestored = true;
		}
		_oxygen = Mathf.Clamp(_oxygen, 0f, _maxOxygen * _player.PlayerSkills.PhysicalMultiplier);
		if (_oxygen >= 0f && !Mathf.Approximately(_oxygen, _maxOxygen * _player.PlayerSkills.PhysicalMultiplier))
		{
			float value2 = _oxygen / (_maxOxygen * _player.PlayerSkills.PhysicalMultiplier);
			OnOxygenUpdated(value2);
		}
		else if (!_oxygenRestored)
		{
			OnOxygenRestored();
			_oxygenRestored = true;
		}
	}

	private void UpdateStamina()
	{
		float num = (_extremeCalorieBurn ? (-5f) : 5f);
		if (_fibres <= 0f || _fats <= 0f || _proteins <= 0f)
		{
			_stamina += num * 0.5f * GameTime.RealDeltaTime;
		}
		else
		{
			_stamina += num * GameTime.RealDeltaTime;
		}
		_stamina = Mathf.Clamp(_stamina, 0f, _maxStamina * _player.PlayerSkills.PhysicalMultiplier);
		if (_stamina == 0f)
		{
			_outOfBreath = true;
		}
		else if (_outOfBreath && _stamina >= _maxStamina * _player.PlayerSkills.PhysicalMultiplier - 0.01f)
		{
			_outOfBreath = false;
		}
		float stamina = _stamina / (_maxStamina * _player.PlayerSkills.PhysicalMultiplier);
		OnStaminaUpdated(stamina);
	}

		_fluids -= 16.8f * sleptTime;
		_fluids = Mathf.Clamp(_fluids, 0f, 700f);
		PlayerEffect healthStatusEffect = GetHealthStatusEffect();
		if (healthStatusEffect.IsValid())
		{

		}
	}

	public void Eat(InteractiveType type, MeatProvenance provenance, int calorieGain, int fibresGain, int fatGain, int proteinsGain, float fluidGain = 0f, float vomitChance = 0f, bool canCauseDiarrhoea = false)
	{
		((fluidGain > (float)calorieGain) ? Drank : Ate)?.Invoke(type, provenance);
		bool flag = (float)UnityEngine.Random.Range(0, 100) < vomitChance;
		if (flag)
		{
			Vomit();
			_fluids -= 100f;
		}
		bool flag2 = canCauseDiarrhoea && CheckDiarrhoea(type);
		if (flag2 && _fluids > 0f)
		{
			Poop();
			_fluids -= 100f;
		}
		if (!(flag | flag2))
		{
			calorieGain = (int)((float)calorieGain * _player.PlayerSkills.CookingMultiplier);
			_calories += calorieGain;
			_fibres += fibresGain;
			_proteins += proteinsGain;
			_fats += fatGain;
			_fluids += fluidGain;
		}
	}

	private bool CheckDiarrhoea(InteractiveType type)
	{
		_foodRecords.TryGetValue(type, out var value);
		if (value != null)
		{
			value.AmountEaten++;
			value.LastTimeEaten = GameTime.Now;
			return value.AmountEaten >= value.AmountSick;
		}
		return false;
	}

	private void Poop(float poopTime = 3f)
	{
		Timing.KillCoroutines(_poopSequence);
		_poopSequence = Timing.RunCoroutine(Poop_Sequence(poopTime));
	}

	private IEnumerator<float> Poop_Sequence(float poopTime)
	{
		CanEat = false;
		yield return Timing.WaitForSeconds(2f);
		++_player.Movement.IsBusy;
		OnPoisoned();
		OnPooped(poopTime);
		yield return Timing.WaitForSeconds(poopTime);
		ApplyStatusEffect(StatusEffect.DIARRHOEA);
		--_player.Movement.IsBusy;
		CanEat = true;
	}

	private void Vomit(float vomitTime = 3f)
	{
		Timing.KillCoroutines(_vomitSequence);
		_vomitSequence = Timing.RunCoroutine(Vomit_Sequence(vomitTime));
	}

		++_player.Movement.IsBusy;

		OnVomited(vomitTime);
		yield return Timing.WaitForSeconds(vomitTime);

		CanEat = true;
	}

	private IEnumerator<float> Knockout_Sequence()
	{
		if (!Invincible && !_isKnockedOut)
		{

			if (_player.Crafter.IsPlacing)

			if (_player.IsOwner)

			}
			PlayerRegistry.CheckAndHandleGameOver();
		}
	}

	public void Revive(Transform RagdollRevivePos)
	{
		KnockedOut?.Invoke(obj: false);
		IsKnockedOut = false;
		Vector3 origin = RagdollRevivePos.position + RagdollRevivePos.TransformDirection(Vector3.up) * 4f;
		Vector3 direction = RagdollRevivePos.TransformDirection(Vector3.down);
		if (Physics.Raycast(origin, direction, out var hitInfo, 10f, (1 << Layers.TERRAIN) | (1 << Layers.TERRAIN_OBJECTS)))
		{
			if (hitInfo.point.y + 0.2f >= RagdollRevivePos.position.y)
			{
				_player.transform.position = new Vector3(RagdollRevivePos.position.x, hitInfo.point.y + 1f, RagdollRevivePos.position.z);
			}
			else
			{
				_player.transform.position = RagdollRevivePos.position;
			}
		}
		else
		{
			_player.transform.position = RagdollRevivePos.position;
		}
		_player.transform.rotation = Quaternion.Euler(_player.transform.rotation.x, RagdollRevivePos.rotation.y, _player.transform.rotation.z);
		_player.transform.localScale = new Vector3(1f, 1f, 1f);
		_player.Movement.ResetChecks();
		_player.Character.PlayingEventAnimation = false;
		_player.PlayerCamera.CameraMovement.ResetCameraTarget();
		Reset_Vitals();
		IsReviving = false;
	}

		}
		if (_fluids < (float)num)

		}
		UpdateVitals();
	}

	public bool ApplyStatusEffect(PlayerEffect effect, bool force = false)
	{
		if (Invincible && !force)
		{
			return false;
		}
		bool flag = _playerEffects.Add(effect);
		if (flag)
		{
			effect.StartedTime = GameTime.Now;
			OnPlayerEffectApplied(effect);
			if (!effect.PositiveEffect)
			{
				EventManager.RaiseEvent(new AddStatsEvent(_player.Id, new Stats
				{
					StatusEffects = 1
				}));
			}
			bool isUnderwater = _player.Movement.IsUnderwater;
			int gender = _player.Gender;
			switch (effect.Name)
			{
			case "SUNBLOCK":
				_speech.Play("STATUS_SUNBLOCK", gender, isUnderwater);
				break;
			case "BOOST_BREATH":
				_speech.Play("STATUS_BREATH", gender, isUnderwater);
				break;
			case "BROKEN_BONES":
				RemoveStatusEffect(StatusEffect.SPLINT);
				_speech.Play("STATUS_BROKEN_BONES", gender, isUnderwater);
				break;
			case "POISON":
				_speech.Play("STATUS_POISON", gender, isUnderwater);
				OnPoisoned();
				break;
			case "ILLNESS":
				_speech.Play("STATUS_ILLNESS", gender, isUnderwater);
				break;
			case "HYPOTHERMIA":
				_speech.Play("STATUS_HYPOTHERMIA", gender, isUnderwater);
				break;
			case "DEHYDRATION":
				_speech.Play("STATUS_DEHYDRATION", gender, isUnderwater);
				break;
			case "STARVING":
				_speech.Play("STATUS_STARVING", gender, isUnderwater);
				break;
			case "BLEEDING":
				_speech.Play("STATUS_BLEEDING", gender, isUnderwater);
				break;
			case "TIRED":
				_speech.Play("STATUS_TIRED", gender, isUnderwater);
				break;
			case "DIARRHOEA":
				_speech.Play("STATUS_DIARRHOEA", gender, isUnderwater);
				break;
			case "MALNUTRITION":
				_speech.Play("STATUS_MALNUTRITION", gender, isUnderwater);
				break;
			}
		}
		return flag;
	}

	public bool RemoveStatusEffect(PlayerEffect effect)
	{
		bool flag = _playerEffects.Remove(effect);
		if (flag)
		{
			OnPlayerEffectRemoved(effect);
		}
		return flag;
	}

	private void RemoveNegativeStatusEffects()
	{
		HashSet<PlayerEffect> hashSet = new HashSet<PlayerEffect>();
		foreach (PlayerEffect playerEffect in _playerEffects)
		{
			if (!playerEffect.PositiveEffect)
			{
				hashSet.Add(playerEffect);
			}
		}
		foreach (PlayerEffect item in hashSet)
		{
			RemoveStatusEffect(item);
		}
	}
