def read_compressed_uint(buf, pos):
    if pos >= len(buf):
        return 0, pos
    b0 = buf[pos]
    if (b0 & 0x80) == 0:
        return b0, pos + 1
    elif (b0 & 0xC0) == 0x80 and pos + 1 < len(buf):
        return ((b0 & 0x3F) << 8) | buf[pos + 1], pos + 2
    elif (b0 & 0xE0) == 0xC0 and pos + 3 < len(buf):
        return ((b0 & 0x1F) << 24) | (buf[pos + 1] << 16) | (buf[pos + 2] << 8) | buf[pos + 3], pos + 4
    return 0, pos + 1

def get_il_slice(data, foff):
    if not foff or foff < 0 or foff >= len(data):
        return "InvalidOffset", b""
    b0 = data[foff]
    if (b0 & 0x03) == 0x02:
        code_size = b0 >> 2
        return f"Tiny ({code_size}B)", data[foff + 1 : foff + 1 + code_size]
    elif (b0 & 0x07) == 0x03:
        if foff + 12 > len(data):
            return "TruncatedFat", b""
        _, max_stack, code_size, _ = struct.unpack_from("<HHII", data, foff)
        return f"Fat (stack={max_stack}, {code_size}B)", data[foff + 12 : foff + 12 + code_size]
    return f"NonIL(0x{b0:02x})", b""
