METHOD: LoadZonePositions
------------------------------------------------------------
LoadZonePositions()
	{
		if (World.GenerationZonePositons == null)
		{
			World.CreateWorldZonePoints(WORLD_SEED);
			Debug.Log("StrandedWorld::CreateZones:: No Zone Positions Loaded - Creating New World Zone Positions");
		}
		for (int i = 0; i < 49; i++)
		{
			Vector3 zero = Vector3.zero;
			zero.x = World.GenerationZonePositons[i].x;
			zero.z = World.GenerationZonePositons[i].y;
			_zones[i].transform.position = zero;
			if (i == 0)
			{
				_zones[i].IsStartingIsland = true;
			}
		}
	}

METHOD: ApplyWorldToZones
------------------------------------------------------------
ApplyWorldToZones()
	{
		LoadZonePositions();
		for (int i = 0; i < 49; i++)
		{
			Zone zone = Zones[i];
			Map map = World.MapList[i];
			zone.Terrain.terrainData.SetHeights(0, 0, map.HeightmapData);
			zone.ZoneName = map.EditorData.Name;
			zone.Id = map.EditorData.Id;
			zone.Version = map.EditorData.VersionNumber;
			zone.Biome = map.EditorData.Biome;
			zone.IsMapEditor = map.IsCustomMap();
			zone.IsUserMap = map.IsUserMap();
			zone.Seed = i;
			Texture2D height = WorldTools.CreateHeightMapTexture(zone.Terrain.terrainData);
			Texture2D shore = WorldTools.CreateShoreMaskTexture(zone.Terrain.terrainData);
			if (zone.Biome == Zone.BiomeType.ISLAND || zone.Biome == Zone.BiomeType.ISLAND_SMALL || zone.Biome == Zone.BiomeType.ISLAND_ROCK || zone.Biome == Zone.BiomeType.CARRIER)
			{
				zone.WaveOverlay = AddWaveOverlay(zone.gameObject, height, shore);
			}
			zone.SaveContainer.transform.SetAsLastSibling();
		}
	}

METHOD: FindEndGameZone
------------------------------------------------------------
FindEndGameZone()
	{
		for (int i = 0; i < Zones.Length; i++)
		{
			if (Zones[i].Id.Equals("INTERNAL_MISSION_3"))
			{
				return Zones[i];
			}
		}
		return null;
	}

METHOD: LoadMap
------------------------------------------------------------
LoadMap(string mapPath, string mapName, bool checkVersion = false)
	{
		using FileStream editorStream = new FileStream(PathTools.Combine(mapPath, mapName + "_EDITOR.map"), FileMode.Open, FileAccess.Read);
		using FileStream objectStream = new FileStream(PathTools.Combine(mapPath, mapName + "_OBJECT.map"), FileMode.Open, FileAccess.Read);
		using FileStream heightStream = new FileStream(PathTools.Combine(mapPath, mapName + "_HEIGHT.map"), FileMode.Open, FileAccess.Read);
		Map map = new Map();
		map.Load(editorStream, objectStream, heightStream, checkVersion);
		return map;
	}

METHOD: SaveMap
------------------------------------------------------------
SaveMap(Map map, string directoryPath)
	{
		string mapFolderName = GetMapFolderName(map);
		string text = PathTools.Combine(directoryPath, mapFolderName);
		if (!Directory.Exists(text))
		{
			Directory.CreateDirectory(text);
		}
		using FileStream streamEditor = new FileStream(PathTools.Combine(text, mapFolderName + "_EDITOR.map"), FileMode.Create, FileAccess.Write, FileShare.ReadWrite);
		using FileStream streamObject = new FileStream(PathTools.Combine(text, mapFolderName + "_OBJECT.map"), FileMode.Create, FileAccess.Write, FileShare.ReadWrite);
		using FileStream streamHeight = new FileStream(PathTools.Combine(text, mapFolderName + "_HEIGHT.map"), FileMode.Create, FileAccess.Write, FileShare.ReadWrite);
		map.Save(streamEditor, streamObject, streamHeight);
	}

METHOD: Compress
------------------------------------------------------------
Compress(this float value)
	{
		if ((double)value > 1.0)
		{
			value = 1f;
		}
		if ((double)value < 0.0)
		{
			value = 0f;
		}
		return (ushort)(value * 65535f);
	}

METHOD: Decompress
------------------------------------------------------------
Decompress(this ushort value)
	{
		return (float)(int)value / 65535f;
	}

METHOD: ToCompressStable
------------------------------------------------------------
ToCompressStable(this ushort value)

		return _toCompressStable[value];
	}
